//+------------------------------------------------------------------+ //| KC_Colored_Candles.mq5 | //| Keltner Channel with Colored Candles | //| | //| Keltner Channel: | //| Middle = EMA(price, EMA Period) default 20 | //| Upper = Middle + Multiplier x ATR(ATR Period) default 20, 1| //| Lower = Middle - Multiplier x ATR(ATR Period) | //| | //| Candle colors (by CLOSE vs. the bands): | //| Close above Upper Band -> Green | //| Close below Lower Band -> Red | //| Close inside the bands -> Gray | //| | //| All periods, multiplier, colors, and line styles are inputs. | //| | //| ORIGINAL PROMPT: | //| Please create a Keltner Channel with colored candles. Default | //| period EMA 20, default ATR 20, default ATR multiplier 1. All | //| candles that close above the upper band to be colored Green and | //| all candles to close below the lower line to be colored red. | //| Candles closing inside the bands to be colored gray. Properties | //| and colors to be modifiable by the user. | //+------------------------------------------------------------------+ #property copyright "KC Colored Candles" #property version "1.00" #property description "Keltner Channel with candles colored by close relative to the bands" #property indicator_chart_window #property indicator_buffers 9 #property indicator_plots 4 //--- Plot 1: Colored candles (Open, High, Low, Close + color index) #property indicator_label1 "KC Candles" #property indicator_type1 DRAW_COLOR_CANDLES #property indicator_color1 clrLime,clrRed,clrGray #property indicator_style1 STYLE_SOLID #property indicator_width1 1 //--- Plot 2: Upper band #property indicator_label2 "KC Upper" #property indicator_type2 DRAW_LINE #property indicator_color2 clrDodgerBlue #property indicator_style2 STYLE_SOLID #property indicator_width2 1 //--- Plot 3: Middle line #property indicator_label3 "KC Middle" #property indicator_type3 DRAW_LINE #property indicator_color3 clrOrange #property indicator_style3 STYLE_DOT #property indicator_width3 1 //--- Plot 4: Lower band #property indicator_label4 "KC Lower" #property indicator_type4 DRAW_LINE #property indicator_color4 clrDodgerBlue #property indicator_style4 STYLE_SOLID #property indicator_width4 1 //+------------------------------------------------------------------+ //| Input Parameters | //+------------------------------------------------------------------+ input group "=== Keltner Channel ===" input int InpEMAPeriod = 20; // EMA Period input int InpATRPeriod = 20; // ATR Period input double InpMultiplier = 1.0; // ATR Multiplier input ENUM_APPLIED_PRICE InpAppliedPrice = PRICE_CLOSE; // EMA Applied Price input group "=== Candle Colors ===" input color InpAboveColor = clrLime; // Close above Upper Band input color InpBelowColor = clrRed; // Close below Lower Band input color InpInsideColor = clrGray; // Close inside the Bands input group "=== Band Lines ===" input bool InpShowBands = true; // Show Keltner lines input color InpUpperColor = clrDodgerBlue; // Upper Band color input color InpMiddleColor = clrOrange; // Middle Line color input color InpLowerColor = clrDodgerBlue; // Lower Band color input ENUM_LINE_STYLE InpBandStyle = STYLE_SOLID; // Upper/Lower line style input ENUM_LINE_STYLE InpMiddleStyle = STYLE_DOT; // Middle line style input int InpBandWidth = 1; // Line width input group "=== Chart ===" input bool InpHideNative = true; // Hide the chart's own candles //+------------------------------------------------------------------+ //| Indicator Buffers | //+------------------------------------------------------------------+ double OpenBuf[]; double HighBuf[]; double LowBuf[]; double CloseBuf[]; double ColorBuf[]; double UpperBuf[]; double MiddleBuf[]; double LowerBuf[]; double ATRBuf[]; // calculation buffer (not plotted) int g_hEMA = INVALID_HANDLE; int g_hATR = INVALID_HANDLE; // Saved chart colors so they can be restored on removal color g_saved_bull = clrNONE; color g_saved_bear = clrNONE; color g_saved_up = clrNONE; color g_saved_down = clrNONE; color g_saved_line = clrNONE; bool g_chart_saved = false; //+------------------------------------------------------------------+ //| Initialization | //+------------------------------------------------------------------+ int OnInit() { if(InpEMAPeriod < 1 || InpATRPeriod < 1) { Print("KC Colored Candles: periods must be >= 1"); return(INIT_PARAMETERS_INCORRECT); } //--- Buffers SetIndexBuffer(0, OpenBuf, INDICATOR_DATA); SetIndexBuffer(1, HighBuf, INDICATOR_DATA); SetIndexBuffer(2, LowBuf, INDICATOR_DATA); SetIndexBuffer(3, CloseBuf, INDICATOR_DATA); SetIndexBuffer(4, ColorBuf, INDICATOR_COLOR_INDEX); SetIndexBuffer(5, UpperBuf, INDICATOR_DATA); SetIndexBuffer(6, MiddleBuf, INDICATOR_DATA); SetIndexBuffer(7, LowerBuf, INDICATOR_DATA); SetIndexBuffer(8, ATRBuf, INDICATOR_CALCULATIONS); //--- Candle colors (user-modifiable) PlotIndexSetInteger(0, PLOT_COLOR_INDEXES, 3); PlotIndexSetInteger(0, PLOT_LINE_COLOR, 0, InpAboveColor); PlotIndexSetInteger(0, PLOT_LINE_COLOR, 1, InpBelowColor); PlotIndexSetInteger(0, PLOT_LINE_COLOR, 2, InpInsideColor); //--- Band line appearance (user-modifiable) PlotIndexSetInteger(1, PLOT_LINE_COLOR, InpUpperColor); PlotIndexSetInteger(2, PLOT_LINE_COLOR, InpMiddleColor); PlotIndexSetInteger(3, PLOT_LINE_COLOR, InpLowerColor); PlotIndexSetInteger(1, PLOT_LINE_STYLE, InpBandStyle); PlotIndexSetInteger(2, PLOT_LINE_STYLE, InpMiddleStyle); PlotIndexSetInteger(3, PLOT_LINE_STYLE, InpBandStyle); for(int p = 1; p <= 3; p++) { PlotIndexSetInteger(p, PLOT_LINE_WIDTH, InpBandWidth); PlotIndexSetInteger(p, PLOT_DRAW_TYPE, InpShowBands ? DRAW_LINE : DRAW_NONE); } //--- Empty values for(int p = 0; p <= 3; p++) PlotIndexSetDouble(p, PLOT_EMPTY_VALUE, 0.0); int draw_begin = MathMax(InpEMAPeriod, InpATRPeriod); for(int p = 0; p <= 3; p++) PlotIndexSetInteger(p, PLOT_DRAW_BEGIN, draw_begin); //--- Indicator handles g_hEMA = iMA(_Symbol, _Period, InpEMAPeriod, 0, MODE_EMA, InpAppliedPrice); g_hATR = iATR(_Symbol, _Period, InpATRPeriod); if(g_hEMA == INVALID_HANDLE || g_hATR == INVALID_HANDLE) { Print("KC Colored Candles: failed to create EMA/ATR handles"); return(INIT_FAILED); } IndicatorSetString(INDICATOR_SHORTNAME, StringFormat("KC Colored Candles (EMA %d, ATR %d, x%.2f)", InpEMAPeriod, InpATRPeriod, InpMultiplier)); IndicatorSetInteger(INDICATOR_DIGITS, _Digits); //--- Hide the chart's native candles so the colored ones are visible if(InpHideNative) { g_saved_bull = (color)ChartGetInteger(0, CHART_COLOR_CANDLE_BULL); g_saved_bear = (color)ChartGetInteger(0, CHART_COLOR_CANDLE_BEAR); g_saved_up = (color)ChartGetInteger(0, CHART_COLOR_CHART_UP); g_saved_down = (color)ChartGetInteger(0, CHART_COLOR_CHART_DOWN); g_saved_line = (color)ChartGetInteger(0, CHART_COLOR_CHART_LINE); g_chart_saved = true; ChartSetInteger(0, CHART_COLOR_CANDLE_BULL, clrNONE); ChartSetInteger(0, CHART_COLOR_CANDLE_BEAR, clrNONE); ChartSetInteger(0, CHART_COLOR_CHART_UP, clrNONE); ChartSetInteger(0, CHART_COLOR_CHART_DOWN, clrNONE); ChartSetInteger(0, CHART_COLOR_CHART_LINE, clrNONE); ChartRedraw(); } return(INIT_SUCCEEDED); } //+------------------------------------------------------------------+ //| Deinitialization | //+------------------------------------------------------------------+ void OnDeinit(const int reason) { if(g_hEMA != INVALID_HANDLE) IndicatorRelease(g_hEMA); if(g_hATR != INVALID_HANDLE) IndicatorRelease(g_hATR); //--- Restore the chart's own candle colors if(g_chart_saved) { ChartSetInteger(0, CHART_COLOR_CANDLE_BULL, g_saved_bull); ChartSetInteger(0, CHART_COLOR_CANDLE_BEAR, g_saved_bear); ChartSetInteger(0, CHART_COLOR_CHART_UP, g_saved_up); ChartSetInteger(0, CHART_COLOR_CHART_DOWN, g_saved_down); ChartSetInteger(0, CHART_COLOR_CHART_LINE, g_saved_line); g_chart_saved = false; ChartRedraw(); } } //+------------------------------------------------------------------+ //| Main Calculation | //+------------------------------------------------------------------+ int OnCalculate(const int rates_total, const int prev_calculated, const datetime &time[], const double &open[], const double &high[], const double &low[], const double &close[], const long &tick_volume[], const long &volume[], const int &spread[]) { int warmup = MathMax(InpEMAPeriod, InpATRPeriod); if(rates_total < warmup) return(0); //--- Make sure the EMA / ATR data is ready if(BarsCalculated(g_hEMA) < rates_total) return(0); if(BarsCalculated(g_hATR) < rates_total) return(0); //--- Copy only what is new (plus the forming bar) int to_copy = (prev_calculated > rates_total || prev_calculated <= 0) ? rates_total : rates_total - prev_calculated + 1; if(CopyBuffer(g_hEMA, 0, 0, to_copy, MiddleBuf) <= 0) return(0); if(CopyBuffer(g_hATR, 0, 0, to_copy, ATRBuf) <= 0) return(0); int start = (prev_calculated <= 0) ? 0 : prev_calculated - 1; for(int i = start; i < rates_total; i++) { //--- Candle OHLC OpenBuf[i] = open[i]; HighBuf[i] = high[i]; LowBuf[i] = low[i]; CloseBuf[i] = close[i]; //--- Warm-up bars: no bands yet, plain gray candle if(i < warmup) { UpperBuf[i] = 0.0; MiddleBuf[i] = 0.0; LowerBuf[i] = 0.0; ColorBuf[i] = 2; continue; } //--- Keltner bands double mid = MiddleBuf[i]; double rng = InpMultiplier * ATRBuf[i]; UpperBuf[i] = mid + rng; LowerBuf[i] = mid - rng; //--- Candle color from the close vs. the bands if(close[i] > UpperBuf[i]) ColorBuf[i] = 0; // above upper -> Green else if(close[i] < LowerBuf[i]) ColorBuf[i] = 1; // below lower -> Red else ColorBuf[i] = 2; // inside -> Gray } return(rates_total); } //+------------------------------------------------------------------+
